### 1. File List & Responsibilities

*   **`adapter.cpp`**: Driver entry point, WDM adapter instantiation, PortCls initialization, subdevice registration, and physical connection binding.
*   **`minwave.h` / `minwave.cpp`**: Implements `IMiniportWaveRT`. Handles filter instantiation, descriptor mapping, and spawning stream objects.
*   **`minwavestream.h` / `minwavestream.cpp`**: Implements `IMiniportWaveRTStream`. Handles cyclic buffer allocation, DMA position reporting, and State changes (calls your `Rp1I2sHwInit`, `Start`, `Stop`).
*   **`mintopo.h` / `mintopo.cpp`**: Implements `IMiniportTopology`. Exposes the physical topology (DAC, Speaker) required by Windows Audio Engine to mount a render endpoint.
*   **`tables.h` / `tables.cpp`**: Contains the `PCFILTER_DESCRIPTOR` definitions, pin arrays, node arrays, and connections.
*   **`minwave.inf`**: Media-class INF file mapping the driver to the hardware and registering KS categories.

### 2. `adapter.cpp` (Adapter Initialization & Start)

```cpp
#include <portcls.h>

// Forward declarations of your miniport creation functions
NTSTATUS CreateMiniportWaveRT(PUNKNOWN* Unknown, REFCLSID ClassId);
NTSTATUS CreateMiniportTopology(PUNKNOWN* Unknown, REFCLSID ClassId);

NTSTATUS StartDevice(PDEVICE_OBJECT DeviceObject, PIRP Irp, PRESOURCELIST ResourceList)
{
    NTSTATUS status;
    PPORT portWave = NULL;
    PPORT portTopo = NULL;
    PUNKNOWN miniportWave = NULL;
    PUNKNOWN miniportTopo = NULL;

    // 1. Create and Init Wave Port/Miniport
    status = PcNewPort(&portWave, CLSID_PortWaveRT);
    if (NT_SUCCESS(status)) {
        status = CreateMiniportWaveRT(&miniportWave, CLSID_MiniportDriverWaveRT); // Implement this to instantiate your class
    }
    if (NT_SUCCESS(status)) {
        status = portWave->Init(DeviceObject, Irp, miniportWave, miniportWave, ResourceList);
    }
    if (NT_SUCCESS(status)) {
        status = PcRegisterSubdevice(DeviceObject, L"Wave", portWave);
    }

    // 2. Create and Init Topology Port/Miniport
    status = PcNewPort(&portTopo, CLSID_PortTopology);
    if (NT_SUCCESS(status)) {
        status = CreateMiniportTopology(&miniportTopo, CLSID_MiniportDriverTopology);
    }
    if (NT_SUCCESS(status)) {
        status = portTopo->Init(DeviceObject, Irp, miniportTopo, miniportTopo, ResourceList);
    }
    if (NT_SUCCESS(status)) {
        status = PcRegisterSubdevice(DeviceObject, L"Topology", portTopo);
    }

    // 3. Register Physical Connection: Wave Bridge Pin (1) -> Topo Bridge Pin (0)
    if (NT_SUCCESS(status)) {
        status = PcRegisterPhysicalConnection(
            DeviceObject, miniportWave, 1, miniportTopo, 0);
    }

    if (portWave) portWave->Release();
    if (portTopo) portTopo->Release();
    if (miniportWave) miniportWave->Release();
    if (miniportTopo) miniportTopo->Release();

    return status;
}

NTSTATUS AddDevice(PDRIVER_OBJECT DriverObject, PDEVICE_OBJECT PhysicalDeviceObject)
{
    return PcAddAdapterDevice(DriverObject, PhysicalDeviceObject, StartDevice, 0, 0);
}

extern "C" NTSTATUS DriverEntry(PDRIVER_OBJECT DriverObject, PUNICODE_STRING RegistryPath)
{
    return PcInitializeAdapterDriver(DriverObject, RegistryPath, AddDevice);
}
```

### 4. Minimal Descriptor Tables (`tables.cpp`)

To expose a render pin, Windows requires an unbroken chain from the Host (OS) to a physical output.

```cpp
#include <portcls.h>

// --- KSDATARANGE_AUDIO ---
static KSDATARANGE_AUDIO PinDataRangesRender[] = {
    {
        {
            sizeof(KSDATARANGE_AUDIO), 0, 0, 0,
            STATICGUIDOF(KSDATAFORMAT_TYPE_AUDIO),
            STATICGUIDOF(KSDATAFORMAT_SUBTYPE_PCM),
            STATICGUIDOF(KSDATAFORMAT_SPECIFIER_WAVEFORMATEX)
        },
        2, 16, 16, 48000, 48000 // Channels, MinBits, MaxBits, MinRate, MaxRate
    }
};

static PKSDATARANGE PinDataRangePointersRender[] = { (PKSDATARANGE)&PinDataRangesRender[0] };

// --- WAVE FILTER ---
// Pin 0: Host Render (IN)
// Pin 1: Bridge Out (OUT) to Topology
static PCPIN_DESCRIPTOR WavePins[] = {
    {
        0, 0, 0, NULL, // MaxInstances...
        {
            0, NULL, 0, NULL,
            KSPIN_DATAFLOW_IN, KSPIN_COMMUNICATION_SINK,
            &KSCATEGORY_AUDIO, &KSAUDFNAME_3D_STEREO,
            0 // KSPIN_FLAG_RENDER? usually implicitly handled by OS categories.
        }
    },
    {
        0, 0, 0, NULL,
        {
            0, NULL, 0, NULL,
            KSPIN_DATAFLOW_OUT, KSPIN_COMMUNICATION_NONE,
            &KSCATEGORY_AUDIO, NULL, 0
        }
    }
};

// Map Pin 0 formats
WavePins[0].KsPinDescriptor.DataRangesCount = 1;
WavePins[0].KsPinDescriptor.DataRanges = PinDataRangePointersRender;

PCFILTER_DESCRIPTOR WaveFilterDescriptor = {
    0, NULL, // AutomationTable
    sizeof(WavePins)/sizeof(PCPIN_DESCRIPTOR), WavePins,
    0, NULL, // Nodes
    0, NULL, // Connections
    0, NULL  // Categories
};

// --- TOPOLOGY FILTER ---
// Pin 0: Bridge In (IN) from Wave
// Pin 1: Physical Endpoint Out (OUT)
static PCPIN_DESCRIPTOR TopoPins[] = {
    { 0,0,0,NULL, { 0,NULL,0,NULL, KSPIN_DATAFLOW_IN, KSPIN_COMMUNICATION_NONE, &KSCATEGORY_AUDIO, NULL, 0 } },
    { 0,0,0,NULL, { 0,NULL,0,NULL, KSPIN_DATAFLOW_OUT, KSPIN_COMMUNICATION_SINK, &KSCATEGORY_AUDIO, &KSNODETYPE_SPEAKER, 0 } }
};

static PCNODE_DESCRIPTOR TopoNodes[] = {
    { 0, NULL, &KSNODETYPE_DAC, NULL }
};

static PCCONNECTION_DESCRIPTOR TopoConnections[] = {
    { PCFILTER_NODE, 0, 0, 1 }, // Topo Pin 0 IN -> DAC Node IN (1)
    { 0, 0, PCFILTER_NODE, 1 }  // DAC Node OUT (0) -> Topo Pin 1 OUT
};

PCFILTER_DESCRIPTOR TopoFilterDescriptor = {
    0, NULL,
    sizeof(TopoPins)/sizeof(PCPIN_DESCRIPTOR), TopoPins,
    sizeof(TopoNodes)/sizeof(PCNODE_DESCRIPTOR), TopoNodes,
    sizeof(TopoConnections)/sizeof(PCCONNECTION_DESCRIPTOR), TopoConnections,
    0, NULL
};
```

### 6. Build Specifics (cl.exe / link.exe)

Since you are bypassing MSBuild, use the following exact flags and linker targets for an ARM64 C++ WDM/PortCls driver:

**Include Paths:**
`-I"C:\Program Files (x86)\Windows Kits\10\Include\10.0.26100.0\km"`
`-I"C:\Program Files (x86)\Windows Kits\10\Include\10.0.26100.0\shared"`

**`cl.exe` Flags:**
```text
/nologo /c /Zi /W4 /WX
/kernel                 ; Enforces kernel C++ rules (no exceptions/no RTTI/pool allocators)
/GR-                    ; Disable RTTI explicitly
/EHs-c-                 ; Disable C++ Exceptions explicitly
/GS                     ; Buffer security checks
/Zc:threadSafeInit-     ; Disable thread-safe statics (not supported in kernel natively without libcntpr)
/arch:armv8.0           ; Target baseline ARM64
/D _ARM64_ /D ARM64 /D STD_CALL /D CONDITION_HANDLING=1 /D NT_UP=1 
/D NT_INST=0 /D WIN32=100 /D _NT1X_=100 /D WINNT=1 
/D _WIN32_WINNT=0x0A00 /D WINVER=0x0A00 /D _WIN32_IE=0x0A00 
/D PC_IMPLEMENTATION
```

**`link.exe` Flags & Libraries:**
```text
/nologo /machine:arm64 /driver /subsystem:native /NODEFAULTLIB 
/ENTRY:DriverEntry      ; WDM standard entry point
/OPT:REF /OPT:ICF
```

**Required Link Libraries:**
*   `ntoskrnl.lib` (Core WDM)
*   `hal.lib` (Hardware Abstraction)
*   `portcls.lib` (Port Class audio framework functions)
*   `stdunk.lib` (Provides `CUnknown` base and `IUnknown` macros implementation)
*   `ksguid.lib` (Provides KS UUIDs like `KSDATAFORMAT_TYPE_AUDIO`, `KSCATEGORY_AUDIO`)
*   `uuid.lib` (Provides standard UUIDs like `CLSID_PortWaveRT`)
*   `BufferOverflowFastFailK.lib` (Resolves `/GS` checks in kernel mode)
*   `libcntpr.lib` (Kernel C++ runtime support for basic new/delete/purecalls)

*(Note on C++ allocators: If `libcntpr.lib` and `stdunk.lib` still leave `operator new` unresolved, implement a dummy global `void* operator new(size_t size) { return ExAllocatePool2(POOL_FLAG_NON_PAGED, size, 'gaTM'); }` inside your code).*
